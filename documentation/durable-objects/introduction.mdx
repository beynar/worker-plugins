## Creating a Durable Object

```typescript
export class MyDurableObject extends createDurableObject() {
  constructor(ctx: DurableObjectState, env: Env) {
    super(ctx, env);
    // Auto-initialized: KV storage, WebSocket manager, Scheduler
  }

  router = {
    // HTTP procedures
  };

  ws_out = {
    // WebSocket output procedures
  };

  ws_in = {
    // WebSocket input handlers
  };

  tasks = {
    // Scheduled tasks
  };
}
```
